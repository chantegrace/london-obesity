--View all records\
SELECT * FROM health_inequalities;\
\
-- Boroughs with highest childhood obesity rates \
SELECT borough, obesity_2024\
FROM health_inequalities\
ORDER BY obesity_2024 DESC;\
\
-- Boroughs with lowest childhood obesity rates\
SELECT borough, obesity_2024\
FROM health_inequalities\
ORDER BY obesity_2024 ASC;\
\
-- Boroughs with highest health inequality rates (most deprivated)\
SELECT borough, average_of_healt\
FROM health_inequalities\
ORDER BY average_of_healt ASC /*the higher the number the less deprivated*/\
LIMIT 5;\
\
-- Boroughs with lowest health inequality rates (least deprivated)\
SELECT borough, average_of_healt\
FROM health_inequalities\
ORDER BY average_of_healt DESC /*the higher the number the less deprivated*/\
LIMIT 5;\
\
-- Boroughs with the largest increase in obesity\
SELECT borough, obesity_change\
FROM health_inequalities\
ORDER BY obesity_change DESC;\
\
-- Boroughs with the lowest increase in obesity\
SELECT borough, obesity_change\
FROM health_inequalities\
ORDER BY obesity_change ASC;\
\
-- Boroughs with obesity above 25%.\
SELECT borough, obesity_2024\
FROM health_inequalities\
WHERE obesity_2024 > 0.25;\
\
-- Boroughs that are improving childhood obesity rates \
SELECT Borough,\
Obesity_Change\
FROM health_inequalities\
WHERE Obesity_Change < 0;\
\
/* Mutliple Columns*/\
SELECT\
Borough,\
average_of_healt,\
Obesity_2024\
FROM health_inequalities\
ORDER BY Obesity_2024 DESC;}
